; STARTUP
; Resets stack, clears memory, draws initial board, and starts attract mode
0089: 31 FA 40    LXI SP, $40FA    ; INIT_STACK <= $40F9 (63 bytes reserved for stack)
008C: CD E4 08    CALL $08E4       ; INIT_GAME
; Sets CURRENT_PIECE to '■' and the MOVE coordinates to (3,2),
; which is the equivalent of CPU_OPENING_MOVE_TABLE(1). However, this
; doesn't get used by anything before being changed in attract mode.
008F: 3E 05       MVI A, #05
0091: 32 8A 40    STA $408A        ; CURRENT_PIECE = '■'
0094: 3E 03       MVI A, #03
0096: 32 8B 40    STA $408B        ; MOVE_COL = 3
0099: 3E 02       MVI A, #02
009B: 32 8C 40    STA $408C        ; MOVE_ROW = 2

009F: 32 FC 40    STA $40FC        ; GAME_SCORED_FLAG = 0
00A2: 3E 07       MVI A, #07       ; ATTRACT_MOVE_COUNT = 7

; The coin selector logic is a bit of a black box, so the following are
; educated guesses based on code behavior. Analysis of the involved
; hardware would be needed to determine exact meanings.
; 
; States tested by the game:
; #0E -> #03 = coin drop with some kind of anti-fraud/debounce test?
; #06 = another accepted coin path, potentially used when the
;       Reset button is pressed while the timer is active?
; #05 = game in progress / coin blocker engaged?
; #0F/#07 = presumed idle state when waiting for coin

; ATTRACT_MODE_LOOP
00A4: F3          DI
00A5: 32 88 40    STA $4088
00A8: 3A 00 A0    LDA $A000        ; READ_COIN_SLOT (possible values: #0E, #05, #06, #03)
00AB: E6 0F       ANI #0F
00AD: FE 0E       CPI #0E
00AF: CA BC 00    JZ $00BC         ; #0E: REREAD_COIN_SLOT
00B2: FE 05       CPI #05
00B4: CA C6 00    JZ $00C6         ; #05: CONTINUE_ATTRACT
00B7: FE 06       CPI #06
00B9: CA 0A 01    JZ $010A         ; #06: COIN_INSERTED

; REREAD_COIN_SLOT
00BC: 3A 00 A0    LDA $A000        ; READ_COIN_SLOT
00BF: E6 07       ANI #07
00C1: FE 03       CPI #03
00C3: CA 0A 01    JZ $010A         ; #03: COIN_INSERTED

; CONTINUE_ATTRACT
; Attract mode toggles messages between "CPU OTHELLO" and "INSERT COIN."
; Every time it toggles to "INSERT COIN," the computer plays a piece.
; There is no sound. After 7 moves are played, the program resets.
00C6: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE
00C9: CD 6F 09    CALL $096F       ; DRAW_MESSAGE
00CC: 00 2B                        ; "ComOthello"
00CE: CD 72 03    CALL $0372       ; LONG_DELAY [1s]
00D1: CD 72 03    CALL $0372       ; LONG_DELAY [1s]
00D4: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE
00D7: CD 6F 09    CALL $096F       ; DRAW_MESSAGE
00DA: 00 3B                        ; "InsertCoin"
00DC: 3A 00 60    LDA $6000        ; READ_INPUT
00DF: FE 7F       CPI #7F          ; Service:
00E1: CA FF 00    JZ $00FF         ; SET_SERVICE_MODE_FLAGS
                                   ; else
; Note that this toggles the piece before it calculates any moves, so
; the side that STARTUP doesn't seed in CURRENT_PIECE will go first
; in attract mode.
00E4: CD 2C 05    CALL $052C       ; TOGGLE_CURRENT_PIECE
00E7: CD 6F 08    CALL $086F       ; CLEAR_MOVE_CURSORS
00EA: CD 90 07    CALL $0790       ; DRAW_GRID
00ED: 3E 01       MVI A, #01
00EF: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 1
; Note how it doesn't use the MOVE coordinates that were set in STARTUP.
; It also can't use FIRST_MOVE_CPU at all, since that is strictly for '■'
; going first. Instead, it simply finds the best move every turn. For
; the first move, since all four of the possible opening moves are
; equivalent, it uses the last one scanned, which is (3,5).
00F2: CD 91 03    CALL $0391       ; CPU_FIND_AND_PLAY_BEST_MOVE
00F5: 3A 88 40    LDA $4088
00F8: 3D          DCR A            ; if --ATTRACT_MOVE_COUNT != 0,
00F9: C2 A4 00    JNZ $00A4        ; ATTRACT_MODE_LOOP
                                   ; else
00FC: C3 00 00    JMP $0000        ; RESET

; SET_SERVICE_MODE_FLAGS
; If Service B is enabled, set CPU_OPENING_MOVE_TABLE_INDEX = 4,
; GAME_MODE = Service, and reenable sound. In this mode, both sides
; are controlled by the CPU and all input prompts are disabled.
00FF: 06 04       MVI B, #04       ; B = 4
0101: AF          XRA A
0102: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 0
0105: 3E 7F       MVI A, #7F       ; A = #7F
0107: C3 56 01    JMP $0156        ; FIRST_MOVE_CPU

; COIN_INSERTED
; Reset attract mode state, reenable sound, and prompt for game
; type selection. Each time through the prompt loop changes what
; the CPU's first move will be if 1P Gote is selected.
010A: 31 FA 40    LXI SP, #40FA    ; INIT_STACK <= $40F9
010D: CD E4 08    CALL $08E4       ; INIT_GAME
0110: AF          XRA A
0111: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 0

; RESET_PSEUDORANDOM_OPENING_MOVE
0114: F3          DI
0115: 06 04       MVI B, #04       ; PSEUDORANDOM_OPENING_MOVE = 4

; PROMPT_SELECT_GAME
0117: F3          DI
0118: C5          PUSH B           ; save PSEUDORANDOM_OPENING_MOVE
0119: CD 6F 09    CALL $096F       ; DRAW_MESSAGE
011C: 00 46                        ; "Pick Game"
011E: C1          POP B            ; restore PSEUDORANDOM_OPENING_MOVE
011F: FB          EI
0120: 3A 00 60    LDA $6000        ; READ_INPUT
; By the rules of Othello, black ('■') always moves first, but that is
; not the case here. 1P is always white ('+') and 2P (or the CPU) is
; always black ('■'). Sente and Gote are terms borrowed from Go. Here
; they indicate which side moves first.
0123: FE FE       CPI #FE          ; 1P Sente:
0125: CA 3E 01    JZ $013E         ; FIRST_MOVE_+
0128: FE FD       CPI #FD          ; 1P Gote:
012A: CA 56 01    JZ $0156         ; FIRST_MOVE_CPU
012D: FE FB       CPI #FB          ; 2P Sente:
012F: CA 3E 01    JZ $013E         ; FIRST_MOVE_+
0132: FE F7       CPI #F7          ; 2P Gote:
0134: CA 4D 01    JZ $014D         ; FIRST_MOVE_■
0137: 05          DCR B            ; if --PSEUDORANDOM_OPENING_MOVE != 0,
0138: C2 17 01    JNZ $0117        ; PROMPT_SELECT_GAME
                                   ; else
013B: C3 14 01    JMP $0114        ; RESET_PSEUDORANDOM_OPENING_MOVE

; Instead of having a distinct turn indicator value, when playing a 2P
; game the GAME_MODE is toggled back and forth between #FB and #F7.
; This gets used to determine which player's inputs to read, where to
; display the on-screen turn indicator, etc.

; FIRST_MOVE_+
013E: F3          DI
013F: 32 FF 40    STA $40FF        ; GAME_MODE = 1P Sente/2P Sente
0142: 3E 03       MVI A, #03       ; CURRENT_PIECE = '+'

; FIRST_MOVE_COMMON
0144: 32 8A 40    STA $408A
0147: CD EC 07    CALL $07EC       ; WAIT_FOR_INPUT_RELEASE
014A: C3 9B 01    JMP $019B        ; PLAYER_TURN_CORE

; FIRST_MOVE_■

014E: 32 FF 40    STA $40FF        ; GAME_MODE = 2P Gote
0151: 3E 05       MVI A, #05       ; CURRENT_PIECE = '■'
0153: C3 44 01    JMP $0144        ; FIRST_MOVE_COMMON

; FIRST_MOVE_CPU
; Input:
;   A = GAME_MODE
;     Service or 1P Gote
;   B = CPU_OPENING_MOVE_INDEX (1..4)
;     is always 4 when GAME_MODE = Service
;
; INIT_GAME is always called here. This is required for the Service
; path because execution reaches this code directly from attract mode.
; For 1P Gote, however, the board was already initialized by
; COIN_INSERTED, making this second INIT_GAME call redundant.
0156: F3          DI
0157: 32 FF 40    STA $40FF        ; GAME_MODE = A
015A: 78          MOV A,B
015B: 32 FE 40    STA $40FE        ; CPU_OPENING_MOVE_INDEX = B
015E: 31 FA 40    LXI SP, $40FA    ; INIT_STACK <= $40F9
0161: CD E4 08    CALL $08E4       ; INIT_GAME
0164: 3E 05       MVI A, #05
0166: 32 8A 40    STA $408A        ; CURRENT_PIECE = '■'
0169: CD C8 08    CALL $08C8       ; PERFORM_CPU_OPENING_MOVE
016C: C3 87 01    JMP $0187        ; EXIT_CPU_TURN

; CPU_TURN

0170: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE

; If both players passed on their previous turns, then neither
; side has a legal move and the game is over.
0176: FE 02       CPI #02          ; if CONSECUTIVE_PASS_COUNTER == 2,
0178: CA C0 01    JZ $01C0         ; WAIT_THEN_JUDGE
                                   ; else
017B: CD 2C 05    CALL $052C       ; TOGGLE_CURRENT_PIECE
017E: CD 6F 08    CALL $086F       ; CLEAR_MOVE_CURSORS
0181: CD C2 0A    CALL $0AC2       ; DRAW_TURN_INDICATOR
0184: CD 91 03    CALL $0391       ; CPU_FIND_AND_PLAY_BEST_MOVE

; EXIT_CPU_TURN

018C: CA 6F 01    JZ $016F         ; CPU_TURN
                                   ; else
; PLAYER_TURN
018F: F3          DI
0190: 3A 89 40    LDA $4089
; If both players passed on their previous turns, then neither
; side has a legal move and the game is over.
0193: FE 02       CPI #02          ; if CONSECUTIVE_PASS_COUNTER == 2,
0195: CA C0 01    JZ $01C0         ; WAIT_THEN_JUDGE
                                   ; else

; PLAYER_TURN_CORE

01A7: 3A FF 40    LDA $40FF

                                   ; else
01B4: C3 6F 01    JMP $016F        ; CPU_TURN

;   A = GAME_MODE

01BD: C3 8F 01    JMP $018F        ; PLAYER_TURN

; WAIT_THEN_JUDGE
; When the caller has detected one of the endgame states,
; the game waits for 1 second and then performs the Judge
; routine. After scoring is complete, the game waits for
; 6 seconds and then automatically resets.
;
; This is a QoL update. The original ROM prompted the
; player to press the Judge button before scoring could
; begin and then prompted the player to press the Reset
; button when scoring was completed.
; 
; Possible endgame states include two consecutive passes
; or every space on the board being filled.
01C0: CD 72 03    CALL $0372       ; LONG_DELAY [1s]

; JUDGE_PRESSED_FINAL
01C3: F3          DI
01C4: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE
01C7: CD 6F 08    CALL $086F       ; CLEAR_MOVE_CURSORS
01CA: CD E5 09    CALL $09E5       ; SCORE_GAME
01CD: CD 6C 03    CALL $036C       ; VERY_LONG_DELAY [6s]
01D0: C3 00 00    JMP $0000        ; RESET

; PLAYER_INPUT_LOOP
; Handles player input during a human player's turn. The Pass and
; Set buttons end the turn when their use is permitted. The Judge
; and Reset buttons trigger a confirmation prompt before ending
; the game immediately. The Arrow buttons update the location of
; the move cursors.
;
; The confirmation prompts are a QoL update that helps prevent
; players from accidentally ending a game in progress.
01D3: F3          DI
01D4: AF          XRA A
01D5: 32 8B 40    STA $408B        ; MOVE_COL = 0
01D8: AF          XRA A
01D9: 32 8C 40    STA $408C        ; MOVE_ROW = 0
01DC: AF          XRA A
01DD: 32 98 40    STA $4098        ; PASS_REQUIRED_FLAG = 0
01E0: CD 3B 08    CALL $083B       ; DRAW_MOVE_CURSORS
01E3: CD 90 07    CALL $0790       ; DRAW_GRID
01E6: CD 1D 08    CALL $081D       ; CHECK_IF_PLAYER_MUST_PASS
; If the player doesn't have any legal moves, then prompt them to
; press the Pass button. Unlike most prompts in the game, this
; doesn't suspend normal input handling. The player can still press
; any button and they will function normally, but since there are
; no legal moves they will be unable to set a piece anywhere.
01E9: 78          MOV A,B
01EA: FE 00       CPI #00          ; if PLAYER_MUST_PASS != 0,
01EC: C2 30 02    JNZ $0230        ; PASS_BUTTON_PROMPT
                                   ; else
; PLAYER_INPUT_LOOP_CORE
01EF: F3          DI
01F0: 3A 98 40    LDA $4098
01F3: FE 01       CPI #01          ; if PASS_REQUIRED_FLAG == 1,
01F5: CA 30 02    JZ $0230         ; PASS_BUTTON_PROMPT
                                   ; else
01F8: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE

; PLAYER_INPUT_LOOP_SKIP_PASS_PROMPT
; In the original ROM, the "PRESS PASS" message was cleared
; and redrawn every iteration of the input loop, causing it
; to flicker. This QoL update keeps the message visible until
; the pass condition is resolved.
01FB: FB          EI
01FC: 3A 00 60    LDA $6000        ; READ_INPUT
; Both players' Pass buttons are mapped to the same value. There's
; no need to differentiate them because the Pass button only has an
; effect when there are no legal moves for the active player.
01FF: FE FB       CPI #FB          ; Pass:
0201: CA 3D 02    JZ $023D         ; PASS_PRESSED
; Both players' Arrow buttons are mapped to the same values. The controls
; are mirrored on opposite sides of the screen, so 'Right' for P1 and
; 'Left' for P2 both move the horizontal cursor in the same direction.
0204: FE FD       CPI #FD          ; Right/Left Arrow:
0206: CA CD 02    JZ $02CD         ; P1_RIGHT_P2_LEFT_PRESSED
; Likewise, 'Down' for P1 and 'Up' for P2 both move the vertical
; cursor in the same direction.
0209: FE FE       CPI #FE          ; Up/Down Arrow:
020B: CA E4 02    JZ $02E4         ; P1_DOWN_P2_UP_PRESSED
; A QoL update adds confirmation prompts when either the
; Judge or Reset buttons are pressed. Pressing the same
; button again confirms the action. Any other button clears
; the prompt and returns control to the player input loop.
020E: FE BF       CPI #BF          ; Judge:
0210: CA 8F 02    JZ $028F         ; CLEAR_AND_ESCAPABLE_PROMPT_FOR_JUDGE
0213: FE DF       CPI #DF          ; Reset:
0215: CA AE 02    JZ $02AE         ; CLEAR_AND_ESCAPABLE_PROMPT_FOR_RESET
0218: 3A FF 40    LDA $40FF        ; else
; Unlike the Pass and Arrow buttons, the Set buttons are treated as
; distinct inputs. The current value of GAME_MODE determines which
; player's Set button is accepted. Both inputs ultimately lead to
; the SET_PRESSED function.
021B: FE FB       CPI #FB          ; if GAME_MODE == 2P Sente,
021D: CA 60 02    JZ $0260         ; CHECK_P1_SET_PRESSED
0220: FE F7       CPI #F7          ; else if GAME_MODE != 2P Gote,
0222: C2 60 02    JNZ $0260        ; CHECK_P1_SET_PRESSED
                                   ; else
0225: 3A 00 60    LDA $6000        ; READ_INPUT
0228: FE F7       CPI #F7          ; P2_Set:
022A: CA 6B 02    JZ $026B         ; SET_PRESSED
                                   ; else
022D: C3 EF 01    JMP $01EF        ; PLAYER_INPUT_LOOP_CORE

; PASS_BUTTON_PROMPT
0230: CD 6F 09    CALL $096F       ; DRAW_MESSAGE
0233: 00 5A                        ; "Must Pass"
0235: 3E 01       MVI A, #01
0237: 32 98 40    STA $4098        ; PASS_REQUIRED_FLAG = 1
023A: C3 FB 01    JMP $01FB        ; PLAYER_INPUT_LOOP_SKIP_PASS_PROMPT

; PASS_PRESSED
023D: F3          DI
023E: CD EC 07    CALL $07EC       ; WAIT_FOR_INPUT_RELEASE
; Run CHECK_IF_PLAYER_MUST_PASS again, even though
; PASS_REQUIRED_FLAG should already contain the result.
;
; If passing is not allowed, a QoL update displays a brief
; "MUST PLAY" message before restarting the player's turn.
; Otherwise, update the pass counter and end the player's turn.
0241: CD 1D 08    CALL $081D       ; CHECK_IF_PLAYER_MUST_PASS
0244: 78          MOV A,B
0245: FE 00       CPI #00          ; if PLAYER_MUST_PASS == 0,
0247: CA 4F 02    JZ $024F         ; CAN'T_PASS
                                   ; else
024A: 21 89 40    LXI #4089
024D: 34          INR M            ; CONSECUTIVE_PASS_COUNTER++

; EXIT_INPUT_LOOP
; Ends the player's current turn.
024E: C9          RET              ; return

; CAN'T_PASS

0255: 00 50                        ; "Must Play"
0257: CD 72 03    CALL $0372       ; LONG_DELAY [1s]

025D: C3 D3 01    JMP $01D3        ; PLAYER_INPUT_LOOP

; CHECK_P1_SET_PRESSED
0260: 3A 00 60    LDA $6000        ; READ_INPUT
0263: FE EF       CPI #EF          ; P1 Set:
0265: CA 6B 02    JZ $026B         ; SET_PRESSED
                                   ; else
0268: C3 EF 01    JMP $01EF        ; PLAYER_INPUT_LOOP_CORE

; SET_PRESSED
026B: F3          DI
; If the attempted move is illegal, a QoL update displays a brief
; "ILLEGAL MOVE" message before restarting the player's turn.
; Otherwise, apply the move and end the player's turn.
026C: CD F5 07    CALL $07F5       ; CHECK_IF_MOVE_IS_LEGAL
026F: 78          MOV A,B
0270: FE 00       CPI #00          ; if MOVE_LEGAL == 0,
0272: CA 7E 02    JZ $027E         ; ILLEGAL_MOVE
                                   ; else
0275: CD F8 03    CALL $03F8       ; PLAY_MOVE_AND_FLIP_OUTFLANKED_PIECES
0278: CD EC 07    CALL $07EC       ; WAIT_FOR_INPUT_RELEASE

; ILLEGAL_MOVE

028C: C3 D3 01    JMP $01D3        ; PLAYER_INPUT_LOOP

; CLEAR_AND_ESCAPABLE_PROMPT_FOR_JUDGE
; Clear the board UI, wait for the original Judge press to be
; released, then ask for confirmation.
028F: CD 6F 08    CALL $086F       ; CLEAR_MOVE_CURSORS
0292: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE
0295: CD EC 07    CALL $07EC       ; WAIT_FOR_INPUT_RELEASE

; ESCAPABLE_PROMPT_FOR_JUDGE

029B: 00 75                        ; "Judge OK?"
029D: FB          EI
029E: 3A 00 60    LDA $6000        ; READ_INPUT
02A1: FE BF       CPI #BF          ; Judge:
02A3: CA C3 01    JZ $01C3         ; JUDGE_PRESSED_FINAL
; Any other button cancels the prompt. Return through
; PLAYER_TURN_CORE so the confirmation message is cleared
; and the turn display is restored.

02A8: C2 9B 01    JNZ $019B        ; PLAYER_TURN_CORE
                                   ; else
02AB: C3 98 02    JMP $0298        ; ESCAPABLE_PROMPT_FOR_JUDGE

; CLEAR_AND_ESCAPABLE_PROMPT_FOR_RESET
; Clear the board UI, wait for the original Reset press to be

02AE: CD 6F 08    CALL $086F       ; CLEAR_MOVE_CURSORS
02B1: CD 5A 09    CALL $095A       ; CLEAR_MESSAGE

; Any other button cancels the prompt. Return through

                                   ; else
02CA: C3 B7 02    JMP $02B7        ; ESCAPABLE_PROMPT_FOR_RESET

; UPDATE_BEST_MOVE_EVAL_SCORE_IF_BETTER
; Input:
;   BC = CANDIDATE_MOVE_EVAL_SCORE

; BEST_MOVE_EVAL_SCORE. Scores are signed 16-bit values.
; If the candidate score is greater than or equal to the

                                   ; else

; GENERATE_CANDIDATE_MOVE_EVAL_SCORE
; Output:
;   BC = CANDIDATE_MOVE_EVAL_SCORE
;
; Scans all of the spaces of the ANALYSIS_BOARD after a
; candidate move has been applied to it and generates a move
; evaluation score for that candidate. The move evaluation
; score uses the values from the BOARD_SPACE_VALUE_TABLE.
; Spaces occupied by the current player's pieces are added
; to the total. Spaces occupied by their opponent's pieces
; are subtracted from the total. Unoccupied spaces are not
; included.
;
; The largest possible value is 2252 (#08CC), so the score
; must be stored in a register pair rather than a single byte.

0648: 23          INX H
0649: 36 00       MVI M, #00       ; CANDIDATE_MOVE_EVAL_SCORE = 0

; GET_SPACE_VALUE
; The BOARD_SPACE_VALUE_TABLE contains only 16 entries. Due to the
; symmetry of the board, every space can be mapped into the upper-left
; 4x4 quadrant, where unique values are stored.
;
; Any ROW or COL coordinate greater than 3 is reflected about the
; board's centerline by replacing it with (7 - coordinate).
;
; The resulting 2-bit ROW and 2-bit COL values are then combined to
; form a 4-bit table index:
;
;     Index = (COL << 2) | ROW
;
; Examples:
;   (3,1) -> (11,01) -> index 1101 = D (13)
;   (5,2) -> (2,2) -> (10,10) -> index 1010 = A (10)
0689: E5          PUSH H           ; save ANALYSIS_BOARD_POINTER
068A: 42          MOV B,D          ; B = ROW_D
068B: CD A6 06    CALL $06A6       ; GET_BOARD_SPACE_VALUE_TABLE_INDEX_BITS
068E: 51          MOV D,C          ; D = BOARD_SPACE_VALUE_TABLE_INDEX_HI_BITS
068F: 43          MOV B,E          ; B = COL_E
0690: CD A6 06    CALL $06A6       ; GET_BOARD_SPACE_VALUE_TABLE_INDEX_BITS
0693: 59          MOV E,C          ; E = BOARD_SPACE_VALUE_TABLE_INDEX_LO_BITS
0694: 7A          MOV A,D
0695: 07          RLC
0696: 07          RLC
0697: 83          ADD E
0698: 5F          MOV E,A          ; E = BOARD_SPACE_VALUE_TABLE_INDEX = (HI_BITS << 2) + LO_BITS
0699: 21 1B 00    LXI H, #001B     ; HL = BOARD_SPACE_VALUE_TABLE(0) = $001B

069D: 85          ADD L
069E: 6F          MOV L,A
069F: 3E 00       MVI A, #00
06A1: 8C          ADC H
06A2: 67          MOV H,A

06A5: C9          RET              ; RETURN

; GET_BOARD_SPACE_VALUE_TABLE_INDEX_BITS
; Input:
;   B = ROW_D or COL_E
; Output:
;   C = coordinate reflected in 0..3
;       0,1,2,3,3,2,1,0
06A6: 3E 03       MVI A, #03
06A8: B8          CMP B            ; if ROW_D|COL_E > 3,
06A9: FA AE 06    JM $06AE         ; MODIFY_INDEX_BITS
                                   ; else
06AC: 48          MOV C,B          ; BITS = ROW_D|COL_E
06AD: C9          RET              ; RETURN

; MODIFY_INDEX_BITS
; Reflects coordinates about the board's centerline.
06AE: 3E 07       MVI A, #07
06B0: 90          SUB B
06B1: 4F          MOV C,A          ; BITS = 7 - ROW_D|COL_E
06B2: C9          RET              ; RETURN

; ADD_VALUE_TO_MOVE_EVAL_SCORE
06B3: CD F9 04    CALL $04F9       ; LOAD_CANDIDATE_MOVE_EVAL_SCORE

; Because a register pair is needed to store the score, the 8-bit
; BOARD_SPACE_VALUE is added to or subtracted from MOVE_EVAL_SCORE
; by repeated increment/decrement of BC. The 8080 has 16-bit addition
; via DAD, but not a compact way to add or subtract an 8-bit value
; directly to/from a register pair.

; INCREMENT_MOVE_EVAL_SCORE
06B6: 03          INX B            ; MOVE_EVAL_SCORE++
06B7: 3D          DCR A            ; if --BOARD_SPACE_VALUE != 0,
06B8: C2 B6 06    JNZ $06B6        ; INCREMENT_MOVE_EVAL_SCORE
                                   ; else
06BB: CD BF 06    CALL $06BF       ; UPDATE_CANDIDATE_MOVE_EVAL_SCORE
06BE: C9          RET              ; RETURN

; UPDATE_CANDIDATE_MOVE_EVAL_SCORE
06BF: E5          PUSH H           ; save ANALYSIS_BOARD_POINTER
06C0: 21 95 40    LXI H, #4095
06C3: 71          MOV M,C
06C4: 23          INX H

; SUB_VALUE_FROM_MOVE_EVAL_SCORE
06C8: CD F9 04    CALL $04F9       ; LOAD_CANDIDATE_MOVE_EVAL_SCORE

; DECREMENT_MOVE_EVAL_SCORE

                                   ; else

; CREATE_DISPLAY_BOARD_MOVE_ASSESSMENT
; Creates a 16-byte packed representation of the board. Each byte
; holds four two-bit values representing a set of four adjacent
; spaces. Empty spaces are scanned to determine whether playing
; there would outflank opponent pieces.
;
; Possible values are:
;   0 = Empty but illegal
;   1 = Empty and legal
;   2 = Contains black piece ('■')
;   3 = Contains white piece ('+')
06D4: 11 00 40    LXI D, #4000     ; DE = DISPLAY_BOARD_POINTER (0,0)
06D7: AF          XRA A
06D8: 32 84 40    STA $4084        ; BOARD_SELECTION_FOR_SCAN = 0 (DISPLAY)
06DB: 21 99 40    LXI H, #4099     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER (0,0)
06DE: 22 92 40    SHLD $4092       ; MOVE_ASSESSMENT_WRITE_POINTER = BOARD_MOVE_ASSESSMENT_POINTER
06E1: 0E 04       MVI C, #04       ; ACCUMULATOR_COUNTER = 4
06E3: 21 00 00    LXI H, #0000     ; ROW_H, COL_L = (0,0)

; ASSESS_SPACE
06E6: 1A          LDAX D
06E7: FE 03       CPI #03          ; if BOARD_POINTER (x,y) == '+'
06E9: CA 2F 07    JZ $072F         ; ASSESSMENT_HANDLE_+
06EC: FE 05       CPI #05          ; if BOARD_POINTER (x,y) == '■'
06EE: CA 34 07    JZ $0734         ; ASSESSMENT_HANDLE_■
                                   ; else
06F1: D5          PUSH D           ; save BOARD_POINTER
06F2: 16 01       MVI D, #01       ; DIRECTION = 1
06F4: C5          PUSH B           ; save CURRENT_SPACE_ACCUMULATOR (unnecessary, uninitialized)
; Unlike when applying moves, this call does not flip pieces.
; It only checks whether the empty space would outflank pieces
; in any direction.
06F5: CD 5A 05    CALL $055A       ; INIT_SCAN_CLOCKWISE_FOR_OUTFLANKED_PIECES_FROM_DIRECTION
06F8: C1          POP B            ; restore CURRENT_SPACE_ACCUMULATOR
06F9: 7A          MOV A,D
06FA: FE 09       CPI #09          ; if DIRECTION == 9,
06FC: CA 39 07    JZ $0739         ; ILLEGAL_SPACE
                                   ; else
06FF: 06 01       MVI B, #01       ; CURRENT_SPACE_ACCUMULATOR = 1
0701: D1          POP D            ; restore BOARD_POINTER

; ACCUMULATOR_HANDLER
; Pack CURRENT_SPACE_ACCUMULATOR into MOVE_ASSESSMENT_ACCUMULATOR.
; After four spaces, write the completed byte and advance
; MOVE_ASSESSMENT_WRITE_POINTER.

0704: FE 04       CPI #04          ; if ACCUMULATOR_COUNTER != 4,
0706: C2 3F 07    JNZ $073F        ; LOAD_ACCUMULATOR
                                   ; else
0709: AF          XRA A            ; TEMP_ACCUMULATOR = 0

; ADD_CURRENT_SPACE_ACCUMULATOR
070A: 80          ADD B            ; TEMP_ACCUMULATOR = TEMP_ACCUMULATOR + CURRENT_SPACE_ACCUMULATOR

070C: C2 45 07    JNZ $0745        ; SHIFT_ACCUMULATOR_BITS_AND_SAVE
                                   ; else
070F: 2A 92 40    LHLD $4092
0712: 77          MOV M,A          ; MOVE_ASSESSMENT(WRITE_POINTER) = TEMP_ACCUMULATOR
0713: 23          INX H
0714: 22 92 40    SHLD $4092       ; MOVE_ASSESSMENT_WRITE_POINTER++
0717: 0E 04       MVI C, #04       ; ACCUMULATOR_COUNTER = 4

; GET_NEXT_ASSESSMENT_SPACE
0719: E1          POP H            ; restore ROW_H, COL_L
071A: 2C          INR L            ; COL_L++

071C: FE 08       CPI #08          ; if COL_L == 8,
071E: CA 25 07    JZ $0725         ; GET_NEXT_ASSESSMENT_ROW
                                   ; else
; ASSESS_NEXT_SPACE
0721: 13          INX D            ; BOARD_POINTER++
0722: C3 E6 06    JMP $06E6        ; ASSESS_SPACE

; GET_NEXT_ASSESSMENT_ROW

                                   ; else

; ASSESSMENT_HANDLE_+

; ASSESSMENT_HANDLE_■

0736: C3 02 07    JMP $0702        ; ACCUMULATOR_HANDLER

; ILLEGAL_SPACE

; LOAD_ACCUMULATOR

; SHIFT_ACCUMULATOR_BITS_AND_SAVE

0746: 07          RLC
0747: 32 94 40    STA $4094        ; MOVE_ASSESSMENT_ACCUMULATOR = TEMP_ACCUMULATOR << 2

; START_NEXT_ASSESSMENT_ROW
0752: 1E 00       MVI E, #00       ; COL_E = 0

; GET_NEXT_MOVE_ASSESSMENT_BYTE
0754: 7E          MOV A,M          ; MOVE_ASSESSMENT_BYTE = MOVE_ASSESSMENT(POINTER)

; EXAMINE_MOVE_ASSESSMENT_BITS
0755: 07          RLC              ; if MOVE_ASSESSMENT_HI_BIT = 1,
0756: DA 7E 07    JC $077E         ; SPACE_OCCUPIED
0759: 07          RLC              ; else
075A: 06 00       MVI B, #00       ; SPACE_CONTENTS = 0 (BLANK)

; ADD_CONTENTS_TO_ANALYSIS_BOARD
075C: D5          PUSH D           ; save ROW_D, COL_E
075D: 4F          MOV C,A          ; CURR_MOVE_ASSESSMENT_BYTE = MOVE_ASSESSMENT_BYTE << 2

077A: 79          MOV A,C          ; MOVE_ASSESSMENT_BYTE = CURR_MOVE_ASSESSMENT_BYTE

; SPACE_OCCUPIED
077E: 07          RLC              ; if MOVE_ASSESSMENT_LO_BIT = 1,
077F: DA 87 07    JC $0787         ; SPACE_IS_+
                                   ; else
0782: 06 05       MVI B, #05       ; SPACE_CONTENTS = '■'

; SPACE_IS_+

; ■ +

; FIRST_PASS_REMOVE_HIGH_RISK_SQUARES
; X-squares are the diagonally corner-adjacent squares:
;   (1,1), (1,6), (6,1), (6,6)
; Othello strategy classifies these as high risk positions to avoid
; because they often allow an opponent to capture a corner.
;
; Zeroes the four X-squares in the MOVE_ASSESSMENT, marking them as
; illegal moves. This prevents the CPU from considering them as
; moves on the first pass.
;
; But this also causes those squares to appear empty when the
; ANALYSIS_BOARD is reconstructed from the MOVE_ASSESSMENT. As a
; result, first-pass evaluation can miss otherwise-legal corner
; captures or other outflanked pieces that depend on actual occupancy
; of an X-square. A better implementation would somehow only mark
; these spaces as illegal if they didn't already contain pieces.
08A0: 06 02       MVI B, #02       ; ROW_LOOP_COUNTER = 2
08A2: 21 9B 40    LXI H, #409B     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER(1,0) = $409B

; REMOVE_HIGH_RISK_FROM_ROW
08A5: 7E          MOV A,M
08A6: E6 CF       ANI #CF

; images respectively. Each row of the image is stored in one byte,
; with the five most-significant bits representing the pixels from

; Input:

;   HL = VRAM_POINTER

; DRAW_IMAGE
; Input:
;   DE = IMAGE_DATA_POINTER
;   HL = VRAM_POINTER
;   C = ROW_COUNTER
;   A = PIXEL_TYPE
;
; Scans the bitmap one row at a time and expands set bits
; into VRAM as PIXEL_TYPE. Clear bits are written as 0.
;
; For example:
;   01110000 (#70)

; DRAW_NEXT_ROW

0928: E5          PUSH H           ; save VRAM_POINTER

; GET_AND_DRAW_NEXT_PIXEL
; Shift next image bit into carry, which is tested with JC.

                                   ; else

                                   ; else

                                   ; else

; SCORE_GAME
; Removes the move cursors and clears the ANALYSIS_BOARD before
; making a three-pass scoring loop.
;
; Pass 1 scans the DISPLAY_BOARD for black pieces ('■'). Each
; black piece increments black's score and is copied into the
; ANALYSIS_BOARD, filling spaces from upper-left to lower-right.
;
; Pass 2 scans for blank spaces and copies them into the next
; available ANALYSIS_BOARD spaces.
;
; Pass 3 scans for white pieces ('+'). Each white piece
; increments white's score and is copied into the next available
; ANALYSIS_BOARD space.
;
; As a result, the ANALYSIS_BOARD ends up sorted into black
; pieces, then blanks, then white pieces.
;
; Each space drawn to the ANALYSIS_BOARD, including blanks,
; is accompanied by the VERY_LOW_TONE.
;
; The final scores are displayed in the message area and a
; winner is determined.
