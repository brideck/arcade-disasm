; STARTUP
; Resets stack, clears memory, draws initial board, and starts attract mode
0081: 31 FA 40    LXI SP, $40FA    ; INIT_STACK <= $40F9 (63 bytes reserved for stack)
0084: CD F6 08    CALL $08F6       ; INIT_GAME
; Sets CURRENT_PIECE to '■' and the MOVE coordinates to (3,2),
; which is the equivalent of CPU_OPENING_MOVE_TABLE(1). However, this
; doesn't get used by anything before being changed in attract mode.
0087: 3E 05       MVI A, #05
0089: 32 8A 40    STA $408A        ; CURRENT_PIECE = '■'
008C: 3E 03       MVI A, #03
008E: 32 8B 40    STA $408B        ; MOVE_COL = 3

0096: AF          XRA A
0097: 32 FC 40    STA $40FC        ; GAME_SCORED_FLAG = 0
009A: 3E 07       MVI A, #07       ; ATTRACT_MOVE_COUNT = 7

; The coin selector logic is a bit of a black box, so the following are
; educated guesses based on code behavior. Analysis of the involved
; hardware would be needed to determine exact meanings.
; 
; States tested by the game:
; #0E -> #03 = coin drop with some kind of anti-fraud/debounce test?
; #06 = another accepted coin path, potentially used when the
;       Reset button is pressed while the timer is active?
; #05 = game in progress / coin blocker engaged?
; #0F/#07 = presumed idle state when waiting for coin

; ATTRACT_MODE_LOOP
009C: F3          DI
009D: 32 88 40    STA $4088
00A0: 3A 00 A0    LDA $A000        ; READ_COIN_SLOT (possible values: #0E, #05, #06, #03)
00A3: E6 0F       ANI #0F
00A5: FE 0E       CPI #0E
00A7: CA B4 00    JZ $00B4         ; #0E: REREAD_COIN_SLOT
00AA: FE 05       CPI #05
00AC: CA BE 00    JZ $00BE         ; #05: CONTINUE_ATTRACT
00AF: FE 06       CPI #06
00B1: CA 04 01    JZ $0104         ; #06: COIN_INSERTED

; REREAD_COIN_SLOT
00B4: 3A 00 A0    LDA $A000        ; READ_COIN_SLOT
00B7: E6 07       ANI #07
00B9: FE 03       CPI #03
00BB: CA 04 01    JZ $0104         ; #03: COIN_INSERTED

; CONTINUE_ATTRACT
; Attract mode toggles messages between "CPU OTHELLO" and "INSERT COIN."
; Every time it toggles to "INSERT COIN," the computer plays a piece.
; There is no sound. After 7 moves are played, the program resets.
00BE: CD 72 09    CALL $0972       ; CLEAR_MESSAGE
00C1: CD 87 09    CALL $0987       ; DRAW_MESSAGE
00C4: 01                           ; unused arg
00C5: 00 2B                        ; "CPU OTHELLO"
00C7: CD 62 03    CALL $0362       ; LONG_DELAY [1s]
00CA: CD 62 03    CALL $0362       ; LONG_DELAY [1s]
00CD: CD 72 09    CALL $0972       ; CLEAR_MESSAGE
00D0: CD 87 09    CALL $0987       ; DRAW_MESSAGE
00D3: 01                           ; unused arg
00D4: 00 3B                        ; "INSERT COIN"
00D6: 3A 00 60    LDA $6000        ; READ_INPUT
00D9: FE 7F       CPI #7F          ; Service:
00DB: CA F9 00    JZ $00F9         ; SET_SERVICE_MODE_FLAGS
                                   ; else
; Note that this toggles the piece before it calculates any moves, so
; the side that STARTUP doesn't seed in CURRENT_PIECE will go first
; in attract mode.
00DE: CD 32 05    CALL $0532       ; TOGGLE_CURRENT_PIECE
00E1: CD 81 08    CALL $0881       ; CLEAR_MOVE_CURSORS
00E4: CD A8 07    CALL $07A8       ; DRAW_GRID
00E7: 3E 01       MVI A, #01
00E9: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 1
; Note how it doesn't use the MOVE coordinates that were set in STARTUP.
; It also can't use FIRST_MOVE_CPU at all, since that is strictly for '■'
; going first. Instead, it simply finds the best move every turn. For
; the first move, since all four of the possible opening moves are
; equivalent, it uses the last one scanned, which is (3,5).
00EC: CD 84 03    CALL $0384       ; CPU_FIND_AND_PLAY_BEST_MOVE
00EF: 3A 88 40    LDA $4088
00F2: 3D          DCR A            ; if --ATTRACT_MOVE_COUNT != 0,
00F3: C2 9C 00    JNZ $009C        ; ATTRACT_MODE_LOOP
                                   ; else
00F6: C3 00 00    JMP $0000        ; RESET

; SET_SERVICE_MODE_FLAGS
; If Service B is enabled, set CPU_OPENING_MOVE_TABLE_INDEX = 4,
; GAME_MODE = Service, and reenable sound. In this mode, both sides
; are controlled by the CPU and all input prompts are disabled.
00F9: 06 04       MVI B, #04       ; B = 4
00FB: AF          XRA A
00FC: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 0
00FF: 3E 7F       MVI A, #7F       ; A = #7F
0101: C3 51 01    JMP $0151        ; FIRST_MOVE_CPU

; COIN_INSERTED
; Reset attract mode state, reenable sound, and prompt for game
; type selection. Each time through the prompt loop changes what
; the CPU's first move will be if 1P Gote is selected.
0104: 31 FA 40    LXI SP, #40FA    ; INIT_STACK <= $40F9
0107: CD F6 08    CALL $08F6       ; INIT_GAME
010A: AF          XRA A
010B: 32 FD 40    STA $40FD        ; ATTRACT_MODE_SUPPRESS_SOUND = 0

; RESET_PSEUDORANDOM_OPENING_MOVE
010E: F3          DI
010F: 06 04       MVI B, #04       ; PSEUDORANDOM_OPENING_MOVE = 4

; PROMPT_SELECT_GAME
0111: F3          DI
0112: C5          PUSH B           ; save PSEUDORANDOM_OPENING_MOVE
0113: CD 87 09    CALL $0987       ; DRAW_MESSAGE
0116: 01                           ; unused arg
0117: 00 47                        ; "SELECT GAME"
0119: C1          POP B            ; restore PSEUDORANDOM_OPENING_MOVE
011A: FB          EI
011B: 3A 00 60    LDA $6000        ; READ_INPUT
; By the rules of Othello, black ('■') always moves first, but that is
; not the case here. 1P is always white ('+') and 2P (or the CPU) is
; always black ('■'). Sente and Gote are terms borrowed from Go. Here
; they indicate which side moves first.
011E: FE FE       CPI #FE          ; 1P Sente:
0120: CA 39 01    JZ $0139         ; FIRST_MOVE_+
0123: FE FD       CPI #FD          ; 1P Gote:
0125: CA 51 01    JZ $0151         ; FIRST_MOVE_CPU
0128: FE FB       CPI #FB          ; 2P Sente:
012A: CA 39 01    JZ $0139         ; FIRST_MOVE_+
012D: FE F7       CPI #F7          ; 2P Gote:
012F: CA 48 01    JZ $0148         ; FIRST_MOVE_■
0132: 05          DCR B            ; if --PSEUDORANDOM_OPENING_MOVE != 0,
0133: C2 11 01    JNZ $0111        ; PROMPT_SELECT_GAME
                                   ; else
0136: C3 0E 01    JMP $010E        ; RESET_PSEUDORANDOM_OPENING_MOVE

; Instead of having a distinct turn indicator value, when playing a 2P
; game the GAME_MODE is toggled back and forth between #FB and #F7.
; This gets used to determine which player's inputs to read, where to
; display the on-screen turn indicator, etc.

; FIRST_MOVE_+
0139: F3          DI
013A: 32 FF 40    STA $40FF        ; GAME_MODE = 1P Sente/2P Sente
013D: 3E 03       MVI A, #03       ; CURRENT_PIECE = '+'

; FIRST_MOVE_COMMON
013F: 32 8A 40    STA $408A
0142: CD FE 07    CALL $07FE       ; WAIT_FOR_INPUT_RELEASE
0145: C3 96 01    JMP $0196        ; PLAYER_TURN_CORE

; FIRST_MOVE_■
0148: F3          DI
0149: 32 FF 40    STA $40FF        ; GAME_MODE = 2P Gote
014C: 3E 05       MVI A, #05       ; CURRENT_PIECE = '■'
014E: C3 3F 01    JMP $013F        ; FIRST_MOVE_COMMON

; FIRST_MOVE_CPU
; Input:
;   A = GAME_MODE
;     Service or 1P Gote
;   B = CPU_OPENING_MOVE_INDEX (1..4)
;     is always 4 when GAME_MODE = Service
;
; INIT_GAME is always called here. This is required for the Service
; path because execution reaches this code directly from attract mode.
; For 1P Gote, however, the board was already initialized by
; COIN_INSERTED, making this second INIT_GAME call redundant.
0151: F3          DI
0152: 32 FF 40    STA $40FF        ; GAME_MODE = A
0155: 78          MOV A,B
0156: 32 FE 40    STA $40FE        ; CPU_OPENING_MOVE_INDEX = B
0159: 31 FA 40    LXI SP, $40FA    ; INIT_STACK <= $40F9
015C: CD F6 08    CALL $08F6       ; INIT_GAME
015F: 3E 05       MVI A, #05
0161: 32 8A 40    STA $408A        ; CURRENT_PIECE = '■'
0164: CD DA 08    CALL $08DA       ; PERFORM_CPU_OPENING_MOVE
0167: C3 82 01    JMP $0182        ; EXIT_CPU_TURN

; CPU_TURN
016A: F3          DI
016B: CD 72 09    CALL $0972       ; CLEAR_MESSAGE
016E: 3A 89 40    LDA $4089
; If both players passed on their previous turns, then neither
; side has a legal move and the game is over.
0171: FE 02       CPI #02          ; if CONSECUTIVE_PASS_COUNTER == 2,
0173: CA BD 01    JZ $01BD         ; PROMPT_FOR_JUDGE_WITH_SERVICE_CHECK
                                   ; else
0176: CD 32 05    CALL $0532       ; TOGGLE_CURRENT_PIECE
0179: CD 81 08    CALL $0881       ; CLEAR_MOVE_CURSORS
017C: CD DB 0A    CALL $0ADB       ; DRAW_TURN_INDICATOR
017F: CD 84 03    CALL $0384       ; CPU_FIND_AND_PLAY_BEST_MOVE

; EXIT_CPU_TURN
0182: 3A FF 40    LDA $40FF
0185: FE 7F       CPI #7F          ; if GAME_MODE == Service,
0187: CA 6A 01    JZ $016A         ; CPU_TURN
                                   ; else
; PLAYER_TURN
018A: F3          DI
018B: 3A 89 40    LDA $4089
; If both players passed on their previous turns, then neither
; side has a legal move and the game is over.
018E: FE 02       CPI #02          ; if CONSECUTIVE_PASS_COUNTER == 2,
0190: CA C5 01    JZ $01C5         ; CLEAR_AND_PROMPT_FOR_JUDGE
                                   ; else
0193: CD 32 05    CALL $0532       ; TOGGLE_CURRENT_PIECE

; PLAYER_TURN_CORE
0196: CD 81 08    CALL $0881       ; CLEAR_MOVE_CURSORS
0199: CD 72 09    CALL $0972       ; CLEAR_MESSAGE
019C: CD DB 0A    CALL $0ADB       ; DRAW_TURN_INDICATOR
019F: CD 04 02    CALL $0204       ; PLAYER_INPUT_LOOP
01A2: 01 C5                        ; JUMP_ADDRESS = CLEAR_AND_PROMPT_FOR_JUDGE
01A4: 3A FF 40    LDA $40FF
01A7: FE FB       CPI #FB          ; if GAME_MODE == 2P Sente,
01A9: CA B4 01    JZ $01B4         ; TOGGLE_2P_GAME_MODE

01AE: CA B4 01    JZ $01B4         ; TOGGLE_2P_GAME_MODE
                                   ; else
01B1: C3 6A 01    JMP $016A        ; CPU_TURN

; TOGGLE_2P_GAME_MODE

;   A = GAME_MODE

01B4: 2F          CMA

01BA: C3 8A 01    JMP $018A        ; PLAYER_TURN

01C2: CA DD 01    JZ $01DD         ; JUDGE_PRESSED_FINAL
                                   ; else

; PLAYER_INPUT_LOOP
; Handles player input during a human player's turn. The Pass and
; Set buttons end the turn when their use is permitted. The Judge
; and Reset buttons end the game immediately. The Arrow buttons
; update the location of the move cursors.
0204: F3          DI
0205: AF          XRA A
0206: 32 8B 40    STA $408B        ; MOVE_COL = 0
0209: AF          XRA A
020A: 32 8C 40    STA $408C        ; MOVE_ROW = 0
020D: AF          XRA A
020E: 32 98 40    STA $4098        ; PASS_REQUIRED_FLAG = 0
0211: CD 4D 08    CALL $084D       ; DRAW_MOVE_CURSORS
0214: CD A8 07    CALL $07A8       ; DRAW_GRID
0217: CD 2F 08    CALL $082F       ; CHECK_IF_PLAYER_MUST_PASS
; If the player doesn't have any legal moves, then prompt them to
; press the Pass button. Unlike most prompts in the game, this
; doesn't suspend normal input handling. The player can still press
; any button and they will function normally, but since there are
; no legal moves they will be unable to set a piece anywhere.
021A: 78          MOV A,B
021B: FE 00       CPI #00          ; if PLAYER_MUST_PASS != 0,
021D: C2 6F 02    JNZ $026F        ; PASS_BUTTON_PROMPT
                                   ; else
; PLAYER_INPUT_LOOP_CORE
0220: F3          DI
0221: 3A 98 40    LDA $4098
0224: FE 01       CPI #01          ; if PASS_REQUIRED_FLAG == 1,
0226: CA 6F 02    JZ $026F         ; PASS_BUTTON_PROMPT
                                   ; else
; PLAYER_INPUT_LOOP_SKIP_PASS_PROMPT
0229: CD 72 09    CALL $0972       ; CLEAR_MESSAGE
022C: FB          EI
022D: 3A 00 60    LDA $6000        ; READ_INPUT
; Both players' Pass buttons are mapped to the same value. There's
; no need to differentiate them because the Pass button only has an
; effect when there are no legal moves for the active player.
0230: FE FB       CPI #FB          ; Pass:
0232: CA 80 02    JZ $0280         ; PASS_PRESSED
; Both players' Arrow buttons are mapped to the same values. The controls
; are mirrored on opposite sides of the screen, so 'Right' for P1 and
; 'Left' for P2 both move the horizontal cursor in the same direction.
0235: FE FD       CPI #FD          ; Right/Left Arrow:
0237: CA BC 02    JZ $02BC         ; P1_RIGHT_P2_LEFT_PRESSED
; Likewise, 'Down' for P1 and 'Up' for P2 both move the vertical
; cursor in the same direction.
023A: FE FE       CPI #FE          ; Up/Down Arrow:
023C: CA D3 02    JZ $02D3         ; P1_DOWN_P2_UP_PRESSED
023F: 3A FC 40    LDA $40FC        ; else
; The Judge button is skipped if GAME_SCORED_FLAG is set. This is
; meaningless in practice, since control never returns to the player
; input loop after SCORE_GAME has begun.
0242: FE 01       CPI #01
0244: CA 4F 02    JZ $024F         ; if GAME_SCORED_FLAG != 1,
0247: 3A 00 60    LDA $6000        ; READ_INPUT
; The Judge and Reset buttons are immediate game-ending actions.
; There is no confirmation prompt.
024A: FE BF       CPI #BF          ; Judge:
024C: CA 96 02    JZ $0296         ; JUDGE_PRESSED
                                   ; else
024F: 3A 00 60    LDA $6000        ; READ_INPUT
0252: FE DF       CPI #DF          ; Reset:
0254: CA 00 00    JZ $0000         ; RESET
0257: 3A FF 40    LDA $40FF        ; else
; Unlike the Pass and Arrow buttons, the Set buttons are treated as
; distinct inputs. The current value of GAME_MODE determines which
; player's Set button is accepted. Both inputs ultimately lead to
; the SET_PRESSED function.
025A: FE FB       CPI #FB          ; if GAME_MODE == 2P Sente,
025C: CA 9E 02    JZ $029E         ; CHECK_P1_SET_PRESSED
025F: FE F7       CPI #F7          ; else if GAME_MODE != 2P Gote,
0261: C2 9E 02    JNZ $029E        ; CHECK_P1_SET_PRESSED
                                   ; else
0264: 3A 00 60    LDA $6000        ; READ_INPUT
0267: FE F7       CPI #F7          ; P2_Set:
0269: CA A9 02    JZ $02A9         ; SET_PRESSED
                                   ; else
026C: C3 20 02    JMP $0220        ; PLAYER_INPUT_LOOP_CORE

; PASS_BUTTON_PROMPT
026F: CD 87 09    CALL $0987       ; DRAW_MESSAGE
0272: 01                           ; unused arg
0273: 00 5F                        ; "PRESS PASS"
0275: CD 68 03    CALL $0368       ; SHORT_DELAY [0.37s]
0278: 3E 01       MVI A, #01
027A: 32 98 40    STA $4098        ; PASS_REQUIRED_FLAG = 1
027D: C3 29 02    JMP $0229        ; PLAYER_INPUT_LOOP_SKIP_PASS_PROMPT

; PASS_PRESSED
0280: F3          DI
0281: CD FE 07    CALL $07FE       ; WAIT_FOR_INPUT_RELEASE
; Run CHECK_IF_PLAYER_MUST_PASS again, even though checking PASS_REQUIRED_FLAG
; should probably suffice. Restart the player's turn if passing is not allowed,
; otherwise update the pass counter and end the player's turn.
0284: CD 2F 08    CALL $082F       ; CHECK_IF_PLAYER_MUST_PASS
0287: 78          MOV A,B
0288: FE 00       CPI #00          ; if PLAYER_MUST_PASS == 0,
028A: CA 04 02    JZ $0204         ; PLAYER_INPUT_LOOP
                                   ; else
028D: 21 89 40    LXI #4089
0290: 34          INR M            ; CONSECUTIVE_PASS_COUNTER++

; EXIT_INPUT_LOOP
; Ends the player's current turn. The caller placed a 2-byte inline
; Judge target after the CALL, so normal exits must skip over it.
0291: E3          XTHL             ; save HL / pop RETURN_ADDRESS
0292: 23          INX H            ; RETURN_ADDRESS++
0293: 23          INX H            ; RETURN_ADDRESS++
0294: E3          XTHL             ; push RETURN_ADDRESS / restore HL
0295: C9          RET              ; return to RETURN_ADDRESS

; JUDGE_PRESSED
; Replace PLAYER_INPUT_LOOP's return address with the inline Judge
; target. Judge is handled outside PLAYER_INPUT_LOOP because it can
; also be invoked from non-input contexts. When called here,
; CLEAR_AND_PROMPT_FOR_JUDGE will not prompt because the Judge
; button has already been pressed and inputs are not cleared, so
; execution falls through to JUDGE_PRESSED_FINAL.
;
; Jumping directly to JUDGE_PRESSED_FINAL instead would appear to
; produce the same behavior.
0296: F3          DI
0297: E3          XTHL             ; save HL / pop RETURN_ADDRESS
0298: 56          MOV D,M          ; D = JUMP_ADDRESS_HI
0299: 23          INX H            ; RETURN_ADDRESS++
029A: 5E          MOV E,M          ; E = JUMP_ADDRESS_LO
029B: EB          XCHG             ; HL = JUMP_ADDRESS
029C: E3          XTHL             ; push JUMP_ADDRESS / restore HL
029D: C9          RET              ; return to JUMP_ADDRESS (CLEAR_AND_PROMPT_FOR_JUDGE)

; CHECK_P1_SET_PRESSED
029E: 3A 00 60    LDA $6000        ; READ_INPUT
02A1: FE EF       CPI #EF          ; P1 Set:
02A3: CA A9 02    JZ $02A9         ; SET_PRESSED
                                   ; else
02A6: C3 20 02    JMP $0220        ; PLAYER_INPUT_LOOP_CORE

; SET_PRESSED
02A9: F3          DI
; Restart the player's turn if the attempted move is illegal, otherwise
; apply the move and end the player's turn.
02AA: CD 07 08    CALL $0807       ; CHECK_IF_MOVE_IS_LEGAL
02AD: 78          MOV A,B
02AE: FE 00       CPI #00          ; if MOVE_LEGAL == 0,
02B0: CA 04 02    JZ $0204         ; PLAYER_INPUT_LOOP
                                   ; else
02B3: CD 07 04    CALL $0407       ; PLAY_MOVE_AND_FLIP_OUTFLANKED_PIECES
02B6: CD FE 07    CALL $07FE       ; WAIT_FOR_INPUT_RELEASE
02B9: C3 91 02    JMP $0291        ; EXIT_INPUT_LOOP

; The move cursors each only move in one direction and wrap around
; at the board edge.

; P1_RIGHT_P2_LEFT_PRESSED
02BC: F3          DI
02BD: 3A 8B 40    LDA $408B        ; A = MOVE_COL
02C0: 3C          INR A            ; A++
02C1: FE 08       CPI #08          ; if A != 8,
02C3: C2 C7 02    JNZ $02C7        ; UPDATE_MOVE_COL

;
; Scans all of the spaces of the ANALYSIS_BOARD after a
; candidate move has been applied to it and generates a move
; evaluation score for that candidate. The move evaluation
; score uses the values from the BOARD_SPACE_VALUE_TABLE.
; Spaces occupied by the current player's pieces are added
; to the total. Spaces occupied by their opponent's pieces
; are subtracted from the total. Unoccupied spaces are not
; included.
;

; must be stored in a register pair rather than a single byte.

; GET_SPACE_VALUE
; The BOARD_SPACE_VALUE_TABLE contains only 16 entries. Due to the
; symmetry of the board, every space can be mapped into the upper-left
; 4x4 quadrant, where unique values are stored.
;
; Any ROW or COL coordinate greater than 3 is reflected about the
; board's centerline by replacing it with (7 - coordinate).
;

; form a 4-bit table index:
;
;     Index = (COL << 2) | ROW
;

;   (3,1) -> (11,01) -> index 1101 = D (13)
;   (5,2) -> (2,2) -> (10,10) -> index 1010 = A (10)

; GET_BOARD_SPACE_VALUE_TABLE_INDEX_BITS

; Output:

;       0,1,2,3,3,2,1,0

; BOARD_SPACE_VALUE is added to or subtracted from MOVE_EVAL_SCORE
; by repeated increment/decrement of BC. The 8080 has 16-bit addition
; via DAD, but not a compact way to add or subtract an 8-bit value
; directly to/from a register pair.

; INCREMENT_MOVE_EVAL_SCORE
06C0: 03          INX B            ; MOVE_EVAL_SCORE++
06C1: 3D          DCR A            ; if --BOARD_SPACE_VALUE != 0,
06C2: C2 C0 06    JNZ $06C0        ; INCREMENT_MOVE_EVAL_SCORE
                                   ; else
06C5: CD C9 06    CALL $06C9       ; UPDATE_CANDIDATE_MOVE_EVAL_SCORE
06C8: C9          RET              ; RETURN

; UPDATE_CANDIDATE_MOVE_EVAL_SCORE
06C9: E5          PUSH H           ; save ANALYSIS_BOARD_POINTER
06CA: 21 95 40    LXI H, #4095
06CD: 71          MOV M,C
06CE: 23          INX H
06CF: 70          MOV M,B          ; CANDIDATE_MOVE_EVAL_SCORE = MOVE_EVAL_SCORE
06D0: E1          POP H            ; restore ANALYSIS_BOARD_POINTER
06D1: C9          RET              ; RETURN

; SUB_VALUE_FROM_MOVE_EVAL_SCORE
06D2: CD FF 04    CALL $04FF       ; LOAD_CANDIDATE_MOVE_EVAL_SCORE

; DECREMENT_MOVE_EVAL_SCORE
06D5: 0B          DCX B            ; MOVE_EVAL_SCORE--
06D6: 3D          DCR A            ; if --BOARD_SPACE_VALUE != 0,
06D7: C2 D5 06    JNZ $06D5        ; DECREMENT_MOVE_EVAL_SCORE
                                   ; else
06DA: CD C9 06    CALL $06C9       ; UPDATE_CANDIDATE_MOVE_EVAL_SCORE
06DD: C9          RET              ; RETURN

; CREATE_DISPLAY_BOARD_MOVE_ASSESSMENT
; Creates a 16-byte packed representation of the board. Each byte
; holds four two-bit values representing a set of four adjacent
; spaces. Empty spaces are scanned to determine whether playing
; there would outflank opponent pieces.
;
; Possible values are:
;   0 = Empty but illegal
;   1 = Empty and legal
;   2 = Contains black piece ('■')
;   3 = Contains white piece ('+')
06DE: 11 00 40    LXI D, #4000     ; DE = DISPLAY_BOARD_POINTER (0,0)
06E1: AF          XRA A
06E2: 32 84 40    STA $4084        ; BOARD_SELECTION_FOR_SCAN = 0 (DISPLAY)
06E5: 21 99 40    LXI H, #4099     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER (0,0)
06E8: C3 F6 06    JMP $06F6        ; CREATE_MOVE_ASSESSMENT_CORE

; CREATE_ANALYSIS_BOARD_MOVE_ASSESSMENT
; An opponent-perspective MOVE_ASSESSMENT is generated here, but
; GENERATE_CANDIDATE_MOVE_EVAL_SCORE only uses the contents of
; ANALYSIS_BOARD directly and does not consult this MOVE_ASSESSMENT.
06EB: 11 40 40    LXI D, #4040     ; DE = ANALYSIS_BOARD_POINTER (0,0)
06EE: 3E 01       MVI A, #01
06F0: 32 84 40    STA $4084        ; BOARD_SELECTION_FOR_SCAN = 1 (ANALYSIS)
06F3: 21 AB 40    LXI H, #40AB     ; HL = ANALYSIS_BOARD_MOVE_ASSESSMENT_POINTER (0,0)

; CREATE_MOVE_ASSESSMENT_CORE
06F6: 22 92 40    SHLD $4092       ; MOVE_ASSESSMENT_WRITE_POINTER = BOARD_MOVE_ASSESSMENT_POINTER
06F9: 0E 04       MVI C, #04       ; ACCUMULATOR_COUNTER = 4
06FB: 21 00 00    LXI H, #0000     ; ROW_H, COL_L = (0,0)

; ASSESS_SPACE
06FE: 1A          LDAX D
06FF: FE 03       CPI #03          ; if BOARD_POINTER (x,y) == '+'
0701: CA 47 07    JZ $0747         ; ASSESSMENT_HANDLE_+
0704: FE 05       CPI #05          ; if BOARD_POINTER (x,y) == '■'
0706: CA 4C 07    JZ $074C         ; ASSESSMENT_HANDLE_■
                                   ; else
0709: D5          PUSH D           ; save BOARD_POINTER
070A: 16 01       MVI D, #01       ; DIRECTION = 1
070C: C5          PUSH B           ; save CURRENT_SPACE_ACCUMULATOR (unnecessary, uninitialized)
; Unlike when applying moves, this call does not flip pieces.
; It only checks whether the empty space would outflank pieces
; in any direction.
070D: CD 60 05    CALL $0560       ; INIT_SCAN_CLOCKWISE_FOR_OUTFLANKED_PIECES_FROM_DIRECTION
0710: C1          POP B            ; restore CURRENT_SPACE_ACCUMULATOR
0711: 7A          MOV A,D
0712: FE 09       CPI #09          ; if DIRECTION == 9,
0714: CA 51 07    JZ $0751         ; ILLEGAL_SPACE
                                   ; else
0717: 06 01       MVI B, #01       ; CURRENT_SPACE_ACCUMULATOR = 1
0719: D1          POP D            ; restore BOARD_POINTER

; ACCUMULATOR_HANDLER
; Pack CURRENT_SPACE_ACCUMULATOR into MOVE_ASSESSMENT_ACCUMULATOR.
; After four spaces, write the completed byte and advance
; MOVE_ASSESSMENT_WRITE_POINTER.
071A: E5          PUSH H           ; save ROW_H, COL_L
071B: 79          MOV A,C
071C: FE 04       CPI #04          ; if ACCUMULATOR_COUNTER != 4,
071E: C2 57 07    JNZ $0757        ; LOAD_ACCUMULATOR
                                   ; else
0721: AF          XRA A            ; TEMP_ACCUMULATOR = 0

; ADD_CURRENT_SPACE_ACCUMULATOR
0722: 80          ADD B            ; TEMP_ACCUMULATOR = TEMP_ACCUMULATOR + CURRENT_SPACE_ACCUMULATOR
0723: 0D          DCR C            ; if --ACCUMULATOR_COUNTER != 0,
0724: C2 5D 07    JNZ $075D        ; SHIFT_ACCUMULATOR_BITS_AND_SAVE
                                   ; else
0727: 2A 92 40    LHLD $4092
072A: 77          MOV M,A          ; MOVE_ASSESSMENT(WRITE_POINTER) = TEMP_ACCUMULATOR
072B: 23          INX H
072C: 22 92 40    SHLD $4092       ; MOVE_ASSESSMENT_WRITE_POINTER++
072F: 0E 04       MVI C, #04       ; ACCUMULATOR_COUNTER = 4

; GET_NEXT_ASSESSMENT_SPACE
0731: E1          POP H            ; restore ROW_H, COL_L
0732: 2C          INR L            ; COL_L++
0733: 7D          MOV A,L
0734: FE 08       CPI #08          ; if COL_L == 8,
0736: CA 3D 07    JZ $073D         ; GET_NEXT_ASSESSMENT_ROW
                                   ; else
; ASSESS_NEXT_SPACE
0739: 13          INX D            ; BOARD_POINTER++
073A: C3 FE 06    JMP $06FE        ; ASSESS_SPACE

; GET_NEXT_ASSESSMENT_ROW
073D: 2E 00       MVI L, #00       ; COL_L = 0
073F: 24          INR H            ; ROW_H++
0740: 7C          MOV A,H
0741: FE 08       CPI #08          ; if ROW_H != 8,
0743: C2 39 07    JNZ $0739        ; ASSESS_NEXT_SPACE
                                   ; else
0746: C9          RET              ; RETURN

; ASSESSMENT_HANDLE_+
0747: 06 03       MVI B, #03       ; CURRENT_SPACE_ACCUMULATOR = 3
0749: C3 1A 07    JMP $071A        ; ACCUMULATOR_HANDLER

; ASSESSMENT_HANDLE_■
074C: 06 02       MVI B, #02       ; CURRENT_SPACE_ACCUMULATOR = 2
074E: C3 1A 07    JMP $071A        ; ACCUMULATOR_HANDLER

; ILLEGAL_SPACE
0751: 06 00       MVI B, #00       ; CURRENT_SPACE_ACCUMULATOR = 0
0753: D1          POP D            ; restore BOARD_POINTER
0754: C3 1A 07    JMP $071A        ; ACCUMULATOR_HANDLER

; LOAD_ACCUMULATOR
0757: 3A 94 40    LDA $4094        ; TEMP_ACCUMULATOR = MOVE_ASSESSMENT_ACCUMULATOR
075A: C3 22 07    JMP $0722        ; ADD_CURRENT_SPACE_ACCUMULATOR

; SHIFT_ACCUMULATOR_BITS_AND_SAVE
075D: 07          RLC
075E: 07          RLC
075F: 32 94 40    STA $4094        ; MOVE_ASSESSMENT_ACCUMULATOR = TEMP_ACCUMULATOR << 2
0762: C3 31 07    JMP $0731        ; GET_NEXT_ASSESSMENT_SPACE

; POPULATE_ANALYSIS_BOARD_FROM_DISPLAY_BOARD_MOVE_ASSESSMENT
0765: 21 99 40    LXI H, #4099     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER
0768: 16 00       MVI D, #00       ; ROW_D = 0

; START_NEXT_ASSESSMENT_ROW
076A: 1E 00       MVI E, #00       ; COL_E = 0

; GET_NEXT_MOVE_ASSESSMENT_BYTE
076C: 7E          MOV A,M          ; MOVE_ASSESSMENT_BYTE = MOVE_ASSESSMENT(POINTER)

; EXAMINE_MOVE_ASSESSMENT_BITS
076D: 07          RLC              ; if MOVE_ASSESSMENT_HI_BIT = 1,
076E: DA 96 07    JC $0796         ; SPACE_OCCUPIED
0771: 07          RLC              ; else
0772: 06 00       MVI B, #00       ; SPACE_CONTENTS = 0 (BLANK)

; ADD_CONTENTS_TO_ANALYSIS_BOARD
0774: D5          PUSH D           ; save ROW_D, COL_E
0775: 4F          MOV C,A          ; CURR_MOVE_ASSESSMENT_BYTE = MOVE_ASSESSMENT_BYTE << 2
0776: EB          XCHG             ; HL = ROW_D, COL_E; DE = MOVE_ASSESSMENT_POINTER
0777: CD D2 04    CALL $04D2       ; GET_SPACE_ON_ANALYSIS_BOARD
077A: 70          MOV M,B          ; ANALYSIS_BOARD (D,E) = SPACE_CONTENTS
077B: EB          XCHG             ; HL = MOVE_ASSESSMENT_POINTER
077C: D1          POP D            ; restore ROW_D, COL_E
077D: 1C          INR E            ; COL_E++
077E: 7B          MOV A,E
077F: FE 04       CPI #04          ; if COL_E == 4,
0781: CA A4 07    JZ $07A4         ; ADVANCE_MOVE_ASSESSMENT_POINTER
0784: FE 08       CPI #08          ; if COL_E != 8,
0786: C2 92 07    JNZ $0792        ; CONTINUE_WITH_MOVE_ASSESSMENT_BYTE
                                   ; else
0789: 14          INR D            ; ROW_D++
078A: 23          INX H            ; MOVE_ASSESSMENT_POINTER++
078B: 7A          MOV A,D
078C: FE 08       CPI #08          ; if ROW_D != 8,
078E: C2 6A 07    JNZ $076A        ; START_NEXT_ASSESSMENT_ROW
                                   ; else
0791: C9          RET              ; RETURN

; CONTINUE_WITH_MOVE_ASSESSMENT_BYTE
0792: 79          MOV A,C          ; MOVE_ASSESSMENT_BYTE = CURR_MOVE_ASSESSMENT_BYTE
0793: C3 6D 07    JMP $076D        ; EXAMINE_MOVE_ASSESSMENT_BITS

; SPACE_OCCUPIED
0796: 07          RLC              ; if MOVE_ASSESSMENT_LO_BIT = 1,
0797: DA 9F 07    JC $079F         ; SPACE_IS_+
                                   ; else
079A: 06 05       MVI B, #05       ; SPACE_CONTENTS = '■'
079C: C3 74 07    JMP $0774        ; ADD_CONTENTS_TO_ANALYSIS_BOARD

; SPACE_IS_+
079F: 06 03       MVI B, #03       ; SPACE_CONTENTS = '+'
07A1: C3 74 07    JMP $0774        ; ADD_CONTENTS_TO_ANALYSIS_BOARD

; ADVANCE_MOVE_ASSESSMENT_POINTER
07A4: 23          INX H            ; MOVE_ASSESSMENT_POINTER++
07A5: C3 6C 07    JMP $076C        ; GET_NEXT_MOVE_ASSESSMENT_BYTE

; DRAW_GRID
; Draws the 9 intersecting vertical and horizontal lines that
; make up the 8x8 gameboard. Each line is 49 pixels in length,
; spanning 8 5-pixel cells and the 9 perpendicular 1-pixel lines.
07A8: 0E 09       MVI C, #09       ; NUM_LINES = 9
07AA: 21 09 CD    LXI H, #CD09     ; HORI_LINE_VRAM_POINTER = $CD09
07AD: E5          PUSH H
07AE: D1          POP D            ; VERT_LINE_VRAM_POINTER = $CD09

; DRAW_NEXT_LINES
07AF: E5          PUSH H           ; save HORI_LINE_VRAM_POINTER
07B0: D5          PUSH D           ; save VERT_LINE_VRAM_POINTER
07B1: 06 31       MVI B, #31       ; GRID_LINE_LENGTH = 49

; DRAW_NEXT_PIXELS
07B3: 3E 01       MVI A, #01       ; PIXEL_TYPE = 1
07B5: 77          MOV M,A          ; HORI_LINE_VRAM(POINTER) = PIXEL_TYPE
07B6: 12          STAX D           ; VERT_LINE_VRAM(POINTER) = PIXEL_TYPE
07B7: 2C          INR L            ; HORI_LINE_VRAM_POINTER++
07B8: 14          INR D            ; VERT_LINE_VRAM_POINTER += #0100
07B9: 05          DCR B            ; if --GRID_LINE_LENGTH != 0,
07BA: C2 B3 07    JNZ $07B3        ; DRAW_NEXT_PIXELS
                                   ; else
07BD: D1          POP D            ; restore VERT_LINE_VRAM_POINTER
07BE: E1          POP H            ; restore HORI_LINE_VRAM_POINTER
; Advance pointers by 6 to next lines, moving past the next 5-pixel cell.
07BF: 7C          MOV A,H
07C0: C6 06       ADI #06
07C2: 67          MOV H,A          ; HORI_LINE_VRAM_POINTER += #0600
07C3: 7B          MOV A,E
07C4: C6 06       ADI #06
07C6: 5F          MOV E,A          ; VERT_LINE_VRAM_POINTER += #0006
07C7: 0D          DCR C            ; if --NUM_LINES != 0,
07C8: C2 AF 07    JNZ $07AF        ; DRAW_NEXT_LINES
                                   ; else
07CB: C9          RET              ; RETURN

; ADD_STARTING_PIECES_TO_BOARDS_AND_DRAW
; Initialize the ANALYSIS and DISPLAY boards to Othello's
; standard starting position, then draw the DISPLAY_BOARD.
; Pieces are written directly into the ANALYSIS_BOARD's RAM
; rather than using GET_SPACE_ON_ANALYSIS_BOARD with
; coordinates.
;
; + ■
; ■ +
07CC: 3E 03       MVI A, #03
07CE: 21 5B 40    LXI H, #405B
07D1: 77          MOV M,A          ; ANALYSIS_BOARD(3,3) = '+'
07D2: 3E 03       MVI A, #03
07D4: 21 64 40    LXI H, #4064
07D7: 77          MOV M,A          ; ANALYSIS_BOARD(4,4) = '+'
07D8: 3E 05       MVI A, #05
07DA: 21 5C 40    LXI H, #405C
07DD: 77          MOV M,A          ; ANALYSIS_BOARD(4,3) = '■'
07DE: 3E 05       MVI A, #05
07E0: 21 63 40    LXI H, #4063
07E3: 77          MOV M,A          ; ANALYSIS_BOARD(3,4) = '■'
07E4: CD EE 04    CALL $04EE       ; COPY_ANALYSIS_BOARD_TO_DISPLAY_BOARD
07E7: CD 0E 05    CALL $050E       ; DRAW_DISPLAY_BOARD
07EA: C9          RET              ; RETURN

; GET_SPACE_ON_DISPLAY_BOARD
; Input:
;   H = H_ROW (0..7)
;   L = L_COL (0..7)
; Output:
;   HL = DISPLAY_BOARD_POINTER (H,L)
;
; H_ROW and L_COL are 3-bit values, so the byte offset is:
;   00rrrccc
; which exactly matches the DISPLAY_BOARD's memory map
07EB: 7C          MOV A,H
07EC: 07          RLC
07ED: 07          RLC
07EE: 07          RLC              ; A = H_ROW << 3
07EF: 85          ADD L
07F0: 6F          MOV L,A          ; OFFSET = H_ROW*8 + L_COL
07F1: 26 40       MVI H, #40       ; HL = $4000 + OFFSET
07F3: C9          RET              ; RETURN

; FLIP_SQUARE_ON_ANALYSIS_BOARD
; Input:
;   HL = H_ROW, L_COL
;
; Set ANALYSIS_BOARD (H_ROW,L_COL) to CURRENT_PIECE. This
; routine does not toggle the existing piece. Callers are
; responsible for ensuring that the space contained an
; opponent piece.
07F4: E5          PUSH H           ; save H_ROW, L_COL
07F5: CD D2 04    CALL $04D2       ; GET_SPACE_ON_ANALYSIS_BOARD
07F8: 3A 8A 40    LDA $408A
07FB: 77          MOV M,A          ; ANALYSIS_BOARD (H_ROW, L_COL) = CURRENT_PIECE
07FC: E1          POP H            ; restore H_ROW, L_COL
07FD: C9          RET              ; RETURN

; WAIT_FOR_INPUT_RELEASE
; Loops until no pressed inputs are detected.
07FE: 3A 00 60    LDA $6000        ; READ_INPUT
0801: FE FF       CPI #FF          ; !No Input:
0803: C2 FE 07    JNZ $07FE        ; WAIT_FOR_INPUT_RELEASE
                                   ; else
0806: C9          RET              ; RETURN

; CHECK_IF_MOVE_IS_LEGAL
0807: 3A 8C 40    LDA $408C
080A: 67          MOV H,A          ; H_ROW = MOVE_ROW
080B: 3A 8B 40    LDA $408B
080E: 6F          MOV L,A          ; L_COL = MOVE_COL

; CHECK_IF_SPACE_IS_LEGAL_MOVE
; Input:
;   H = H_ROW (0..7)
;   L = L_COL (0..7)
; Output:
;   B = MOVE_LEGAL (0/1)
;
; A move is legal only if the selected DISPLAY_BOARD space is empty
; and placing CURRENT_PIECE there would outflank at least one
; opponent piece.
080F: E5          PUSH H           ; save H_ROW, L_COL
0810: CD EB 07    CALL $07EB       ; GET_SPACE_ON_DISPLAY_BOARD
0813: 7E          MOV A,M          ; SPACE_CONTENTS = DISPLAY_BOARD(H_ROW,L_COL)
0814: E1          POP H            ; restore H_ROW, L_COL
0815: FE 00       CPI #00          ; if SPACE_CONTENTS != 0,
0817: C2 2C 08    JNZ $082C        ; ILLEGAL_MOVE
081A: AF          XRA A
081B: 32 84 40    STA $4084        ; BOARD_SELECTION_FOR_SCAN = 0 (DISPLAY)
081E: 16 01       MVI D, #01       ; DIRECTION = 1
; Unlike when applying moves, this call does not flip pieces.
; It only checks whether the empty space would outflank pieces
; in any direction. If it would, then MOVE_LEGAL is set to 1.
0820: CD 60 05    CALL $0560       ; INIT_SCAN_CLOCKWISE_FOR_OUTFLANKED_PIECES_FROM_DIRECTION
0823: 7A          MOV A,D
0824: FE 09       CPI #09          ; if DIRECTION == 9,
0826: CA 2C 08    JZ $082C         ; ILLEGAL_MOVE
                                   ; else
0829: 06 01       MVI B, #01       ; MOVE_LEGAL = 1
082B: C9          RET              ; RETURN

; ILLEGAL_MOVE
082C: 06 00       MVI B, #00       ; MOVE_LEGAL = 0
082E: C9          RET              ; RETURN

; CHECK_IF_PLAYER_MUST_PASS
; Output:
;   B = PLAYER_MUST_PASS (0/1)
;
; Scans the entire board for a legal move for the current
; player. The search terminates immediately if one is found.
; If no legal move exists, PLAYER_MUST_PASS is set to 1.
082F: 26 07       MVI H, #07       ; H_ROW = 7

; CHECK_NEXT_ROW
0831: 2E 07       MVI L, #07       ; L_COL = 7

; CHECK_NEXT_SPACE
0833: CD 0F 08    CALL $080F       ; CHECK_IF_SPACE_IS_LEGAL_MOVE
0836: 78          MOV A,B
0837: FE 00       CPI #00          ; if MOVE_LEGAL != 0,
0839: C2 47 08    JNZ $0847        ; PLAYER_HAS_LEGAL_MOVE
083C: 2D          DCR L            ; if --L_COL >= 0,
083D: F2 33 08    JP $0833         ; CHECK_NEXT_SPACE
                                   ; else
0840: 25          DCR H            ; if --H_ROW >= 0,
0841: F2 31 08    JP $0831         ; CHECK_NEXT_ROW
                                   ; else
0844: 06 01       MVI B, #01       ; PLAYER_MUST_PASS = 1
0846: C9          RET              ; RETURN

; PLAYER_HAS_LEGAL_MOVE
0847: 06 00       MVI B, #00       ; PLAYER_MUST_PASS = 0
0849: C9          RET              ; RETURN

; HIGH_TONE_AND_DRAW_MOVE_CURSORS
084A: CD 0D 09    CALL $090D       ; HIGH_TONE

; DRAW_MOVE_CURSORS
; Clear the old move cursors, then draw new 5-pixel length
; cursors matching MOVE_COL/MOVE_ROW.
;
; From 1P's perspective, the horizontal cursor is drawn
; above the board, and the vertical cursor is drawn to the
; left of the board.
084D: CD 81 08    CALL $0881       ; CLEAR_MOVE_CURSORS
0850: 21 0A CA    LXI H, #CA0A     ; HL = VRAM_HORI_CURSOR_POINTER(0) = $CA0A
0853: 3A 8C 40    LDA $408C
0856: 57          MOV D,A          ; D = MOVE_ROW
0857: 3A 8B 40    LDA $408B
085A: 5F          MOV E,A          ; E = MOVE_COL

; SET_HORIZONTAL_CURSOR
085B: 1D          DCR E            ; if --MOVE_COL < 0,
085C: FA 66 08    JM $0866         ; DRAW_HORIZONTAL_CURSOR
                                   ; else
085F: 7D          MOV A,L
; Cursor moves 6 pixels at a time to skip over 5-pixel cell
; and 1-pixel line.
0860: C6 06       ADI #06          ; VRAM_HORI_CURSOR_POINTER += #0006
0862: 6F          MOV L,A          ; VRAM_HORI_CURSOR_POINTER(++)
0863: C3 5B 08    JMP $085B        ; SET_HORIZONTAL_CURSOR

; DRAW_HORIZONTAL_CURSOR
0866: 3A 8A 40    LDA $408A        ; A = CURRENT_PIECE
0869: CD C8 08    CALL $08C8       ; DRAW_HORIZONTAL_MOVE_CURSOR

086C: 21 06 CE    LXI H, #CE06     ; HL = VRAM_VERT_CURSOR_POINTER(0) = $CE06

; SET_VERTICAL_CURSOR
086F: 15          DCR D            ; if --MOVE_ROW < 0,
0870: FA 7A 08    JM $087A         ; DRAW_VERTICAL_CURSOR
                                   ; else
0873: 7C          MOV A,H
; Cursor moves 6 pixels at a time to skip over 5-pixel cell
; and 1-pixel line.
0874: C6 06       ADI #06          ; VRAM_VERT_CURSOR_POINTER += #0600
0876: 67          MOV H,A          ; VRAM_VERT_CURSOR_POINTER(++)
0877: C3 6F 08    JMP $086F        ; SET_VERTICAL_CURSOR

; DRAW_VERTICAL_CURSOR
087A: 3A 8A 40    LDA $408A        ; A = CURRENT_PIECE
087D: CD D1 08    CALL $08D1       ; DRAW_VERTICAL_MOVE_CURSOR
0880: C9          RET              ; RETURN

; CLEAR_MOVE_CURSORS
; Blanks out the entire 52-pixel horizontal and vertical lines
; where the move cursors are displayed.
;
; The cursor lines extend slightly beyond the 49-pixel board span.
; The extra pixels are never drawn as cursors, but clearing them
; simplifies the implementation.
0881: 06 34       MVI B, #34       ; MOVE_CURSOR_LINE_LENGTH = 52
0883: 21 06 CA    LXI H, $CA06     ; VRAM_HORI_CURSOR_POINTER = $CA06
0886: E5          PUSH H
0887: D1          POP D            ; VRAM_VERT_CURSOR_POINTER = $CA06

; BLANK_NEXT_PIXELS
0888: AF          XRA A            ; PIXEL_TYPE = 0 (BLANK)
0889: 77          MOV M,A          ; VRAM_HORI_CURSOR(POINTER) = PIXEL_TYPE
088A: 12          STAX D           ; VRAM_VERT_CURSOR(POINTER) = PIXEL_TYPE
088B: 2C          INR L            ; VRAM_HORI_CURSOR_POINTER++
088C: 14          INR D            ; VRAM_VERT_CURSOR_POINTER += #0100
088D: 05          DCR B            ; if --MOVE_CURSOR_LINE_LENGTH != 0,
088E: C2 88 08    JNZ $0888        ; BLANK_NEXT_PIXELS
                                   ; else
0891: C9          RET              ; RETURN

; FLASH_BOARD_CHANGE
; Alternates between the DISPLAY_BOARD (before the move) and
; ANALYSIS_BOARD (after the move) 3 times in rapid succession,
; highlighting the newly played piece or any resulting flips.
;
; If not in attract mode, LOW_TONE is emitted each time the
; ANALYSIS_BOARD is drawn.
0892: 3E 03       MVI A, #03       ; FLASH_LOOP_COUNTER = 3

; FLASH_BOARD
0894: F5          PUSH PSW         ; save FLASH_LOOP_COUNTER
0895: CD 0E 05    CALL $050E       ; DRAW_DISPLAY_BOARD
0898: CD 76 03    CALL $0376       ; CORE_DELAY [30ms]
089B: 3A FD 40    LDA $40FD
089E: FE 01       CPI #01          ; if ATTRACT_MODE_SUPPRESS_SOUND == 0,
08A0: CA A6 08    JZ $08A6         ; SWAP_BOARD
                                   ; else
08A3: CD 08 09    CALL $0908       ; LOW_TONE

; SWAP_BOARD
08A6: CD 08 05    CALL $0508       ; DRAW_ANALYSIS_BOARD
08A9: CD 76 03    CALL $0376       ; CORE_DELAY [30ms]
08AC: F1          POP PSW          ; restore FLASH_LOOP_COUNTER

; FIRST_PASS_REMOVE_HIGH_RISK_SQUARES
; X-squares are the diagonally corner-adjacent squares:
;   (1,1), (1,6), (6,1), (6,6)
; Othello strategy classifies these as high risk positions to avoid
; because they often allow an opponent to capture a corner.
;
; Zeroes the four X-squares in the MOVE_ASSESSMENT, marking them as
; illegal moves. This prevents the CPU from considering them as
; moves on the first pass.
;
; But this also causes those squares to appear empty when the
; ANALYSIS_BOARD is reconstructed from the MOVE_ASSESSMENT. As a
; result, first-pass evaluation can miss otherwise-legal corner
; captures or other outflanked pieces that depend on actual occupancy
; of an X-square. A better implementation would somehow only mark
; these spaces as illegal if they didn't already contain pieces.
08B2: 06 02       MVI B, #02       ; ROW_LOOP_COUNTER = 2
08B4: 21 9B 40    LXI H, #409B     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER(1,0) = $409B

; REMOVE_HIGH_RISK_FROM_ROW

08BA: 77          MOV M,A          ; DISPLAY_BOARD_MOVE_ASSESSMENT(x,1) = 0
08BB: 23          INX H            ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER++
08BC: 7E          MOV A,M
08BD: E6 F3       ANI #F3
08BF: 77          MOV M,A          ; DISPLAY_BOARD_MOVE_ASSESSMENT(x,6) = 0
08C0: 21 A5 40    LXI H, #40A5     ; HL = DISPLAY_BOARD_MOVE_ASSESSMENT_POINTER(6,0) = $40A5
08C3: 05          DCR B            ; if --ROW_LOOP_COUNTER != 0,
08C4: C2 B7 08    JNZ $08B7        ; REMOVE_HIGH_RISK_FROM_ROW
                                   ; else
08C7: C9          RET              ; RETURN

; DRAW_NEXT_HORI_PIXEL

; PERFORM_CPU_OPENING_MOVE
; Look up the coordinates of the selected CPU_OPENING_MOVE
; from the four-entry table and play it. All of the opening
; moves would be equivalent if using the normal MOVE_ASSESSMENT
; approach, so CPU_FIND_AND_PLAY_BEST_MOVE would just result in
; the same move always being selected.
;
; Instead, the PROMPT_SELECT_GAME loop supplies a pseudo-random

; images respectively. Each row of the image is stored in one byte,
; with the five most-significant bits representing the pixels from

; DRAW_GAME_PIECE
; Input:
;   A = SPACE_CONTENTS
;   HL = VRAM_POINTER
;
; Draws a 5x5 game piece image at the specified VRAM location.

; DRAW_IMAGE
; Input:
;   DE = IMAGE_DATA_POINTER
;   HL = VRAM_POINTER
;   C = ROW_COUNTER
;   A = PIXEL_TYPE
;
; Scans the bitmap one row at a time and expands set bits
; into VRAM as PIXEL_TYPE. Clear bits are written as 0.
;
; For example:
;   01110000 (#70)
; becomes:
;   0 PIXEL_TYPE PIXEL_TYPE PIXEL_TYPE 0
093A: 32 81 40    STA $4081        ; IMAGE_PIXEL_VALUE = PIXEL_TYPE

; DRAW_NEXT_ROW
093D: 06 05       MVI B, #05       ; PIXEL_COUNTER = 5
093F: 1A          LDAX D           ; A = IMAGE_DATA(POINTER)
0940: E5          PUSH H           ; save VRAM_POINTER

; GET_AND_DRAW_NEXT_PIXEL
; Shift next image bit into carry, which is tested with JC.
0941: 17          RAL              ; IMAGE_DATA << 1
0942: F5          PUSH PSW         ; save IMAGE_DATA
                                   ; if IMAGE_DATA_HI_BIT == 1,
0943: DA 66 09    JC $0966         ; USE_IMAGE_PIXEL_VALUE
                                   ; else
0946: 3E 00       MVI A, #00       ; PIXEL_TYPE = #00

; DRAW_NEXT_PIXEL

094A: 23          INX H            ; VRAM_POINTER++
094B: 05          DCR B            ; if --PIXEL_COUNTER != 0,
094C: C2 41 09    JNZ $0941        ; GET_AND_DRAW_NEXT_PIXEL
                                   ; else
094F: 13          INX D            ; IMAGE_DATA_POINTER++
0950: E1          POP H            ; restore VRAM_POINTER
0951: 24          INR H            ; VRAM_POINTER += #0100
0952: 0D          DCR C            ; if --ROW_COUNTER != 0,
0953: C2 3D 09    JNZ $093D        ; DRAW_NEXT_ROW
                                   ; else

0957: D1          POP D            ; restore DE
0958: C1          POP B            ; restore BC
0959: C9          RET              ; RETURN

; SCORE_GAME
; Removes the move cursors and clears the ANALYSIS_BOARD before
; making a three-pass scoring loop.
;
; Pass 1 scans the DISPLAY_BOARD for black pieces ('■'). Each
; black piece increments black's score and is copied into the
; ANALYSIS_BOARD, filling spaces from upper-left to lower-right.
;
; Pass 2 scans for blank spaces and copies them into the next
; available ANALYSIS_BOARD spaces.
;
; Pass 3 scans for white pieces ('+'). Each white piece
; increments white's score and is copied into the next available
; ANALYSIS_BOARD space.
;
; As a result, the ANALYSIS_BOARD ends up sorted into black
; pieces, then blanks, then white pieces.
;
; Each space drawn to the ANALYSIS_BOARD, including blanks,
; is accompanied by the VERY_LOW_TONE.
;
; The final scores are displayed in the message area and a
; winner is determined.
09FE: CD 81 08    CALL $0881       ; CLEAR_MOVE_CURSORS
0A01: AF          XRA A
0A02: 32 82 40    STA $4082        ; ■_COUNT = 0
0A05: AF          XRA A
0A06: 32 83 40    STA $4083        ; +_COUNT = 0
0A09: 3E 01       MVI A, #01
0A0B: 32 FC 40    STA $40FC        ; GAME_SCORED_FLAG = 1

                                   ; else

;   3 = Black ('■'), 2 = Blanks, 1 = White ('+'), 0 = done

; START_SCORING_PASS

0A41: C2 7C 0A    JNZ $0A7C        ; GET_NEXT_SCORE_SPACE
                                   ; else

; CHECK_PLAYER_WIN
; In 1-player games, checks whether 1P defeated the CPU.
; If so, play the victory jingle. Ties, CPU wins, and all
; 2-player games return without playing anything.

                                   ; else
